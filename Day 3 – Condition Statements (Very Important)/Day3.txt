    console.log("Eligible");

    console.log("Not Eligible");
}

🟠 3️⃣ if – else if – else
let marks = 85;

if (marks >= 90) {
    console.log("Grade A");
} else if (marks >= 75) {
    console.log("Grade B");
} else if (marks >= 50) {
    console.log("Grade C");
} else {
    console.log("Fail");
}

🔵 4️⃣ Comparison + Logical Practice

    console.log("Allowed inside");

🟣 5️⃣ Nested if
let username = "admin";
let password = "1234";

if (username === "admin") {
    if (password === "1234") {
        console.log("Login Success");
    } else {
        console.log("Wrong Password");
    }
} else {
    console.log("Wrong Username");
}
