🟡 2️⃣ Example – Simple Calculator using switch

let num2 = 5;
let operator = "+";

switch(operator) {
    case "+":
        console.log(num1 + num2);
        break;
    case "-":
        console.log(num1 - num2);
        break;
    case "*":
        console.log(num1 * num2);
        break;
          case "/":
        console.log(num1 / num2);
        break;
    default:
        console.log("Invalid Operator");
}
