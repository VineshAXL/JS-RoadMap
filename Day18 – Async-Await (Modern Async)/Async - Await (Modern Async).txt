🟡 2️⃣ Basic Example
function getData() {
    return new Promise((resolve) => {
        setTimeout(() => {
            resolve("Data received 📦");
        }, 2000);
    });
}

async function fetchData() {
    let result = await getData();
    console.log(result);
}

fetchData();

👉 await → Promise complete aagum vara wait pannum

🟠 3️⃣ Without async (Old way)
getData().then(result => console.log(result));

👉 async/await → cleaner 💯

🔵 4️⃣ Error Handling (try...catch 🔥)
function checkLogin() {
    return new Promise((resolve, reject) => {
        let success = false;

        if (success) {
            resolve("Login Success");
        } else {
            reject("Login Failed");
        }
    });
}

async function login() {
    try {
        let result = await checkLogin();
        console.log(result);
    } catch (error) {
        console.log(error);
    }
}

login();
🟣 5️⃣ Multiple Await
function step1() {
    return new Promise(res => setTimeout(() => res("Step 1"), 1000));
}

function step2() {
    return new Promise(res => setTimeout(() => res("Step 2"), 1000));
}

async function process() {
    let a = await step1();
    console.log(a);

    let b = await step2();
    console.log(b);
}

process();
