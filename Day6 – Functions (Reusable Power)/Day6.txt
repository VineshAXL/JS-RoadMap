⏰ Time: 2–3 Hours
🎯 Goal: Code reuse panna theriyanum

🟢 1️⃣ What is Function?

👉 Reusable block of code

function greet() {
    console.log("Hello Vinesh 🔥");
}

greet(); // function call

🟡 2️⃣ Function with Parameters
function greet(name) {
    console.log("Hello " + name);
}

greet("Vinesh");
greet("Kumar");
🟠 3️⃣ Function with Return
function add(a, b) {
    return a + b;
}

let result = add(10, 5);
console.log(result);

🟠 3️⃣ Function with Return
function add(a, b) {
    return a + b;
}

let result = add(10, 5);
console.log(result);

🟣 5️⃣ Default Parameter
function greet(name = "Guest") {
    console.log("Hello " + name);
}

greet();
greet("Vinesh");
