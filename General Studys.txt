Let's start with JavaScript from the beginning 😊

What is JavaScript?
JavaScript is a high-level, dynamic, and interpreted programming language used for client-side scripting on the web. It's used to create interactive web pages, web applications, and mobile applications.

Setting up JavaScript
You can write JavaScript code in a .js file and run it using a browser's console or a Node.js environment.

Basic Syntax

// Variables
let name = "John";
const PI = 3.14;
var age = 25;

// Data Types
// String, Number, Boolean, Null, Undefined, Object, Array

// Operators
// Arithmetic: +, -, *, /, %, **
// Comparison: ==, !=, ===, !==, >, <, >=, <=
// Logical: &&, ||, !


Variables and Data Types

let x = 10; // Number
let y = "Hello"; // String
let z = true; // Boolean
let a = null; // Null
let b; // Undefined
let obj = { name: "John", age: 25 }; // Object
let arr = [1, 2, 3]; // Array


Control Structures

// If-Else
if (x > 10) {
  console.log("x is greater than 10");
} else {
  console.log("x is less than or equal to 10");
}

// Switch
let day = "Monday";
switch (day) {
  case "Monday":
    console.log("Today is Monday");
    break;
  default:
    console.log("Today is not Monday");
}

// Loops
for (let i = 0; i < 5; i++) {
  console.log(i);
}

let i = 0;
while (i < 5) {
  console.log(i);
  i++;
}


Functions

function greet(name) {
  console.log(`Hello, ${name}!`);
}

greet("John");

let add = (a, b) => a + b;
console.log(add(2, 3));


Arrays

let numbers = [1, 2, 3, 4, 5];
console.log(numbers[0]); // 1
numbers.push(6);
console.log(numbers); // [1, 2, 3, 4, 5, 6]


Objects

let person = { name: "John", age: 25 };
console.log(person.name); // John
person.country = "USA";
console.log(person); // { name: "John", age: 25, country: "USA" }
